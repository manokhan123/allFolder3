const http = require("http");

const server = http.createServer((req, res) => {
  if (req.url == "/") {
    res.writeHead("HOME PAGE");
    res.end();
  } else if (req.url == "/product") {
    res.writeHead("PRODUCT  PAGE");
    res.end();
  } else if (req.url == "/about") {
    res.writeHead("ABOUT  PAGE");
    res.end();
  } else {
    res.writeHead("404 ERROR Not Found!");
    res.end();
  }
});

const port = 3003;

server.listen(port, () => {
  console.log("Our server is starting now");
});
